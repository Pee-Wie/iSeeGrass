CREATE DATABASE IF NOT EXISTS iseagrass
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE iseagrass;

CREATE TABLE IF NOT EXISTS seagrass_species (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    species_key VARCHAR(100) NOT NULL UNIQUE,
    common_name VARCHAR(150),
    scientific_name VARCHAR(150) NOT NULL UNIQUE,
    genus VARCHAR(100) NOT NULL,
    family VARCHAR(100) NOT NULL,
    region VARCHAR(255),
    habitat TEXT,
    depth_range VARCHAR(100),
    leaf_shape VARCHAR(255),
    leaf_length VARCHAR(100),
    color_description VARCHAR(150),
    iucn_status VARCHAR(50),
    status_color VARCHAR(10),
    description TEXT,
    fun_fact TEXT,
    tags JSON,
    philippine_regions JSON,
    is_philippine_species BOOLEAN NOT NULL DEFAULT FALSE,
    is_cnn_supported BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP
);

-- ============================================================
-- Filter dropdowns
-- ============================================================

CREATE OR REPLACE VIEW v_seagrass_families AS
SELECT DISTINCT family
FROM seagrass_species
ORDER BY family;

CREATE OR REPLACE VIEW v_seagrass_genera AS
SELECT DISTINCT genus
FROM seagrass_species
ORDER BY genus;

CREATE OR REPLACE VIEW v_philippine_seagrass_species AS
SELECT *
FROM seagrass_species
WHERE is_philippine_species = TRUE;

CREATE OR REPLACE VIEW v_cnn_supported_species AS
SELECT *
FROM seagrass_species
WHERE is_cnn_supported = TRUE;

-- Verify imported rows
SELECT COUNT(*) AS catalog_species_count
FROM seagrass_species;

SELECT family, COUNT(*) AS species_count
FROM seagrass_species
GROUP BY family
ORDER BY family;
